# Ростов-Елена-Тур — полный handoff проекта

> Этот файл предназначен для передачи проекта другой нейросети или разработчику. Он содержит контекст, архитектуру, инструкции по запуску и полный исходный код текстовых файлов. Данные о турах и цены в прототипе тестовые.

## 1. Что это за проект

Современный демонстрационный сайт турагентства «Ростов-Елена-Тур»:

- Адаптивный премиальный лендинг для мобильных устройств, планшетов и десктопов;
- Универсальный интеллектуальный движок подбора туров по ВСЕМ направлениям России и мира (Золотое Кольцо, Переславль-Залесский, Суздаль, Байкал, Алтай, Карелия, Калининград, Кавказ, Сочи, Казань, Самара, Турция, Египет, ОАЭ, Бали, Мальдивы, Таиланд, Куба, Хайнань, Европа и любые другие локации);
- AI-турагент «Алиса» на базе Groq (`openai/gpt-oss-20b`) со строгим запретом на фразы вроде «к сожалению» и автоматической генерацией 2–3 проверенных отелей, точных маршрутов, программ питания, фотогалерей и прозрачного расчета цен;
- Интерактивные карточки туров прямо внутри чата с кнопками «Подробнее и рейсы» и «Забронировать»;
- Полноэкранный модальный просмотр презентации отеля (`openTourDetailModal`) с табами («Маршрут и трансфер», «Отель и территория», «Питание», «Что включено»), фотогалереей и прямой поддержкой deep-linking по URL (`#tour=...`);
- 1-click бронирование тура с автозаполнением темы в форму захвата лидов (`#leadModal`);
- Публичный фронтенд размещён на GitHub Pages;
- Защищённый серверный API-прокси размещён на Vercel Functions без раскрытия API ключей.

## 3. Важные правила и ограничения прототипа

- Туры, отели, цены, даты, отзывы и статистика являются демонстрационными/тестовыми данными.
- Диалог в AI-чате сохраняет контекст (состав путешественников, город вылета, ночи, бюджет) и не повторяет приветствия («Здравствуйте») в последующих репликах.
- Никогда не использовать Puter и никогда не вставлять `GROQ_API_KEY` во фронтенд или в публичный репозиторий.
- При любых правках JS проверять синтаксис через `node --check app.js`.
- При обновлении `app.js` или `styles.css` обновлять cache-buster версий в `index.html` (`app.js?v=...`).

## 4. Архитектура и файлы

1. `index.html` — разметка лендинга, карточки, фильтры настроения, форма поиска, модальные окна, AI-чат.
2. `styles.css` — дизайн-система, типографика, адаптивность, карточки предложений в чате, фотогалерея, табы.
3. `app.js` — логика приложения, `DESTINATIONS_CATALOG`, универсальный экстрактор `resolveDestination()`, генератор интерактивных карточек `generateUniversalTourProposal()`, модальные окна и интеграция с Groq.
4. `api/ai.js` — безопасная Serverless-функция на Vercel для взаимодействия с Groq API.
5. `vercel.json` — конфигурация таймаутов Vercel.

const ALLOWED_ORIGINS = new Set([

export default async function handler(req, res) {
  const origin = req.headers.origin || '';
  if (ALLOWED_ORIGINS.has(origin) || origin.endsWith('.e2b.app') || origin.includes('localhost') || origin.includes('127.0.0.1')) {
    res.setHeader('Access-Control-Allow-Origin', origin);
    res.setHeader('Vary', 'Origin');
  } else {
    res.setHeader('Access-Control-Allow-Origin', '*');
  }
  res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
  res.setHeader('Access-Control-Max-Age', '86400');

  if (req.method !== 'POST') return res.status(405).json({ error: 'Method not allowed' });
  if (!process.env.GROQ_API_KEY) return res.status(503).json({ error: 'AI is not configured' });

  let payload = req.body;
  if (typeof payload === 'string') {

      payload = JSON.parse(payload);
    } catch {
      payload = {};

    '5. Главные фишки (спа, бассейн, близость к озеру/морю/достопримечательностям, экскурсии)',

    'Отвечай живым, дружелюбным, профессиональным языком эксперта по туризму.'
  ].join('\n');

  try {
    const upstream = await fetch('https://api.groq.com/openai/v1/chat/completions', {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${process.env.GROQ_API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: 'openai/gpt-oss-20b',
        messages: [
          { role: 'system', content: system },
          ...history
            .filter(item => item && ['user', 'assistant'].includes(item.role) && typeof item.content === 'string')
            .map(item => ({ role: item.role, content: item.content.slice(0, 1200) })),
          { role: 'user', content: message }
        ],
        reasoning_effort: 'low',
        temperature: 0.45,
        max_completion_tokens: 1200
      })
    });

    const data = await upstream.json();
    if (!upstream.ok) {
      console.error('Groq error', upstream.status, data?.error?.message || 'unknown');
      return res.status(502).json({ error: 'Groq request failed' });
    }

    const answer = data?.choices?.[0]?.message?.content?.trim();
    if (!answer) return res.status(502).json({ error: 'Empty Groq response' });

    return res.status(200).json({
      answer,
      provider: 'groq',
      model: 'openai/gpt-oss-20b'
    });
  } catch (error) {
    console.error('AI function error', error?.message || error);
    return res.status(500).json({ error: 'AI request failed' });
  }
}

function applyPhoneMask(input) {
  if (!input) return;
  input.addEventListener('input', () => {
    let val = input.value.replace(/\D/g, '');
    if (val.startsWith('7') || val.startsWith('8')) val = val.slice(1);
    val = val.slice(0, 10);
    let formatted = '+7 ';
    if (val.length > 0) formatted += '(' + val.slice(0, 3);
    if (val.length >= 3) formatted += ') ' + val.slice(3, 6);
    if (val.length >= 6) formatted += '-' + val.slice(6, 8);
    if (val.length >= 8) formatted += '-' + val.slice(8, 10);
    input.value = val.length === 0 ? '' : formatted;
  });
}

// Helper to Clean and Extract Custom Destination Queries
function cleanDestinationQuery(userQuery) {
  let s = ' ' + userQuery.toLowerCase() + ' ';
  const stopWords = [
    'найди мне тур', 'найди тур', 'подобрать тур', 'хочу тур', 'хочу поехать', 'хочу в', 'хочу на',
    'посоветуй отель', 'посоветуй тур', 'куда поехать', 'горящие туры', 'горящий тур',
    'на двоих', 'на троих', 'для двоих', 'для семьи', 'с детьми', 'с ребенком', 'вдвоем', 'вдвоём', 'на одного',
    'на выходные', 'на уикенд', 'все включено', 'всё включено', 'all inclusive', 'с бассейном',
    'в сентябре', 'в октябре', 'в ноябре', 'в декабре', 'в январе', 'в феврале', 'в марте', 'в апреле', 'в мае', 'в июне', 'в июле', 'в августе',
    'летом', 'зимой', 'весной', 'осенью',
    'из москвы', 'из ростова', 'из спб', 'из петербурга', 'из сочи', 'из минеральных вод', 'из минвод',
    'найди', 'тур', 'туры', 'путевка', 'путевки', 'отель', 'отели', 'поездка', 'отдых', 'путешествие',
    'билеты', 'ночей', 'ночи', 'ночь', 'дней', 'дня', 'день', 'неделю', 'недели',
    'хочу', 'посоветуй', 'подобрать', 'ищу', 'поехали', 'отпуск',
    'рублей', 'руб', 'бюджет', 'недорого', 'премиум', 'люкс', 'спа', 'spa',
    'в', 'на', 'по', 'до', 'от', 'для', 'с', 'со', 'под', 'из'
  ];

  s = s.replace(/\d+/g, ' ');
  s = s.replace(/[«»""?!.,;:()\-–—]/g, ' ');

  stopWords.forEach(w => {
    const reg = new RegExp('(^|\\s+)' + w + '(\\s+|$)', 'gi');
    while (reg.test(s)) {
      s = s.replace(reg, ' ');
    }
  });

  let clean = s.replace(/\s+/g, ' ').trim();
  if (clean.length < 2) clean = 'Выбранный курорт';
  return clean.split(' ').map(w => w.charAt(0).toUpperCase() + w.slice(1)).join(' ');
}

// Universal Smart Tour Proposal Generator
function generateUniversalTourProposal(userQuery, isFirstTurn) {
  const ctx = window.travelContext;
  ctx.turnCount = isFirstTurn ? 1 : ctx.turnCount + 1;
  const lower = userQuery.toLowerCase();

  // Resolve Destination
  const dest = resolveDestination(userQuery);
  ctx.lastDestination = dest.locationTitle;

  // Detect Departure City
  if (/из ростов|вылет.*ростов|ростова/.test(lower)) { ctx.departureCity = 'Ростова-на-Дону'; ctx.depCode = 'ROV'; }
  else if (/из москв|вылет.*москв|москвы/.test(lower)) { ctx.departureCity = 'Москвы'; ctx.depCode = 'SVO'; }
  else if (/из соч|вылет.*сочи/.test(lower)) { ctx.departureCity = 'Сочи'; ctx.depCode = 'AER'; }
  else if (/мин.*вод|минвод/.test(lower)) { ctx.departureCity = 'Минеральных Вод'; ctx.depCode = 'MRV'; }
  else if (/питер|санкт-петербург/.test(lower)) { ctx.departureCity = 'Санкт-Петербурга'; ctx.depCode = 'LED'; }

  // Detect Travelers Count
  if (/на троих|3\s*чел/.test(lower)) ctx.adults = 3;
  else if (/на одного|1\s*чел/.test(lower)) ctx.adults = 1;
  else if (/на двоих|2\s*взрос|вдвоем|вдвоём/.test(lower)) ctx.adults = 2;

  if (/с дет|с ребен|ребенок|дети/.test(lower)) {
    const cm = lower.match(/(\d+)\s*(?:реб|дет)/);
    ctx.children = cm ? parseInt(cm[1]) : 1;
  }

  // Detect Duration
  const nightsRangeMatch = lower.match(/(\d+)\s*[-–—]\s*(\d+)\s*(?:ноч|дн)/);
  if (nightsRangeMatch) {
    ctx.nights = Math.round((parseInt(nightsRangeMatch[1]) + parseInt(nightsRangeMatch[2])) / 2);
  } else {
    const singleNightsMatch = lower.match(/(\d+)\s*(?:ноч|дн)/);
    if (singleNightsMatch) ctx.nights = parseInt(singleNightsMatch[1]);
  }

  // Detect Month
  const months = ['январе', 'феврале', 'марте', 'апреле', 'мае', 'июне', 'июле', 'августе', 'сентябре', 'октябре', 'ноябре', 'декабре'];
  months.forEach(m => {
    if (lower.includes(m.slice(0, 4))) ctx.month = m;
  });

  // Detect Budget
  const budgetMatch = lower.match(/(?:до|бюджет\D{0,10})(\d[\d\s]{3,})/);
  if (budgetMatch) ctx.budget = parseInt(budgetMatch[1].replace(/\s/g, ''));

  // Natural Human Travel Agent Intro Speech (NO repetitive "Здравствуйте"!)
  let agentIntro = '';
  if (ctx.turnCount <= 1) {
    agentIntro = `Здравствуйте! С удовольствием помогу организовать поездку. По вашему запросу подготовила отличные проверенные варианты в <b>${dest.locationTitle}</b> (${ctx.adults} взр.${ctx.children ? ' + ' + ctx.children + ' реб.' : ''}, выезд/вылет из ${ctx.departureCity}):`;
  } else {
    agentIntro = `Отличная идея с направлением <b>${dest.locationTitle}</b>! Подобрала лучшие варианты отелей с комфортным размещением для вашего состава (${ctx.adults} взр., на ${ctx.nights} ночей, из ${ctx.departureCity}):`;
  }

  // Generate Hotel Proposals with Thematic High-Resolution Photos
  const photoSet = THEMATIC_PHOTOS[dest.theme] || THEMATIC_PHOTOS.lake;
  const proposals = [];

  dest.hotels.forEach((hotelTemplate, idx) => {
    const tourId = `tour-prop-${Date.now().toString().slice(-5)}-${idx + 1}`;
    const depDay = 5 + idx * 2;
    const startDateStr = `${depDay < 10 ? '0' + depDay : depDay} ${ctx.month.slice(0, 3)} 2026`;
    const endDateStr = `${depDay + ctx.nights} ${ctx.month.slice(0, 3)} 2026`;

    let calculatedPrice = dest.basePrice * (ctx.adults + ctx.children * 0.65) * (ctx.nights / 7) + (idx * 12000);
    if (ctx.budget && calculatedPrice > ctx.budget) {
      calculatedPrice = ctx.budget - (idx * 5000);
    }
    calculatedPrice = Math.round(calculatedPrice / 500) * 500;
    if (calculatedPrice < 24000) calculatedPrice = 24000 + idx * 6000;

    const oldPrice = Math.round((calculatedPrice * 1.12) / 500) * 500;
    const perPerson = Math.round(calculatedPrice / (ctx.adults + ctx.children));

    const tourObj = {
      id: tourId,
      hotelName: hotelTemplate.name,
      stars: hotelTemplate.stars,
      rating: hotelTemplate.rating || '4.8',
      reviewsCount: 280 + idx * 70,
      country: dest.country,
      resort: hotelTemplate.resort || dest.region,
      beach: hotelTemplate.beach || 'Оборудованная территория отдыха',
      food: hotelTemplate.food || dest.foodDefault,
      room: hotelTemplate.room || 'Улучшенный номер (32 м²)',
      departureCity: ctx.departureCity,
      departureAirport: ctx.depCode,
      destinationAirport: 'Курорт',
      destinationAirportName: dest.locationTitle,
      datesText: `${ctx.nights} ночей · ${startDateStr} — ${endDateStr}`,
      datesShort: `${startDateStr} (на ${ctx.nights} ночей)`,
      nights: ctx.nights,
      airline: dest.transportType,
      airlineReturn: dest.transportType,
      flightHours: dest.transportRoute,
      priceTotal: calculatedPrice.toLocaleString('ru-RU') + ' ₽',
      priceOld: oldPrice.toLocaleString('ru-RU') + ' ₽',
      pricePerPerson: perPerson.toLocaleString('ru-RU') + ' ₽ / чел.',
      discount: '-12%',
      features: hotelTemplate.features,
      photos: photoSet,
      description: hotelTemplate.desc,
      adults: ctx.adults,
      children: ctx.children,
      publicLink: `https://reklamadomodedovo-star.github.io/turyrostova-prototype/?tour=${tourId}`
    };

    window.toursRegistry[tourId] = tourObj;
    proposals.push(tourObj);
  });

  return {
    agentIntro: agentIntro,
    destName: dest.locationTitle,
    departureCity: ctx.departureCity,
    dates: `${ctx.nights} ночей в ${ctx.month}`,
    travelersText: `${ctx.adults} взр.${ctx.children ? ' + ' + ctx.children + ' реб.' : ''}`,
    expertTip: dest.expertAdvice,
    proposals: proposals
  };
}

const aiHistory = [];

// Groq API Caller with Dynamic Universal Fallback
async function groqAiReply(text) {
  const isFirstTurn = aiHistory.length === 0;
  const proposalData = generateUniversalTourProposal(text, isFirstTurn);

  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), 12000);

  try {
    const response = await fetch('https://turyrostova-groq-api-reklamadomodedovo-7709.vercel.app/api/ai', {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=UTF-8' },
      body: JSON.stringify({ message: text, history: aiHistory.slice(-8) }),
      signal: controller.signal
    });

    if (!response.ok) throw new Error('Groq network error');
    const data = await response.json();
    if (data && data.answer) {
      aiHistory.push({ role: 'user', content: text }, { role: 'assistant', content: data.answer });
      
      // Extract commentary from Groq
      let groqCommentary = data.answer.split(/(?:\d+[\.\)]\s+)/)[0].trim();
      if (!isFirstTurn) {
        groqCommentary = groqCommentary.replace(/^(?:Здравствуйте|Добрый день|Приветствую)[!.,\s]*/i, '');
      }
      if (groqCommentary && groqCommentary.length > 25 && !groqCommentary.includes('к сожалению')) {
        proposalData.agentIntro = groqCommentary;
      }
    }
  } catch (error) {
    console.warn('Groq live fallback:', error);
    aiHistory.push({ role: 'user', content: text }, { role: 'assistant', content: proposalData.agentIntro });
  } finally {
    clearTimeout(timer);
  }

  return { type: 'smart', data: proposalData };
}

// Send Message Handler

  appendMessage(text, 'user');

  $('small', pending).textContent = now();
  const chatBody = $('#chatBody');
  if (chatBody) chatBody.scrollTop = chatBody.scrollHeight;
}
